# 第六章：型別安全程式碼生成

> 本章說明為何型別安全埋點是可觀測性品質的根基，並提供完整的 Go 與 Python 業務邏輯整合範例，包含 metric 初始化、span 屬性設定的正確寫法與常見錯誤對照。

---

## 6.1 為什麼需要型別安全？

### 危險的手動埋點（容易出錯）

```go
// Go — 手動字串，拼字錯誤無法被編譯器發現
span.SetAttributes(
    attribute.String("pay.order_id", orderID),      // 錯！應該是 payment.order_id
    attribute.String("paymet.provider", "stripe"),  // 錯！拼字錯誤
    attribute.String("payment.statas", "success"),  // 錯！應該是 payment.status
)
```

```python
# Python — 同樣的問題
span.set_attributes({
    "pay.order_id": order_id,       # 錯！
    "paymet.provider": "stripe",    # 錯！拼字錯誤
    "payment.statas": "success",    # 錯！
})
```

這些錯誤的共同特點：**在執行時才發現，而且可能進到生產環境**。Grafana dashboard 上沒有資料、警報規則永遠不觸發，但程式還是正常運作，根本不知道出問題了。

### 安全的生成程式碼（錯誤被編譯期攔截）

```go
// Go — 使用生成的常數，IDE 自動補齊，拼字錯誤 = 編譯失敗
span.SetAttributes(
    semconv.PAYMENT_ORDER_ID.String(orderID),
    semconv.PAYMENT_PROVIDER.String(provider),
    semconv.PAYMENT_STATUS.String("success"),
)
```

### 型別安全的核心價值

| 問題 | 手打字串 | 用生成的常數 |
|------|---------|------------|
| 拼字錯誤 | 跑到生產才發現 | import / 編譯期就報錯 |
| 改名同步 | 靜默出錯，散落各處 | 重新 generate，所有呼叫端編譯失敗，強迫一起更新 |
| 查文件 | 每次都要翻 YAML | IDE hover 顯示 brief、stability、requirement_level |
| Code Review | 很難發現命名錯誤 | 屬性名稱一定正確，Review 只需關注業務邏輯 |

## 6.3 兩類常數的完整說明

### `semconv_attrs` — Span 屬性 Key

Go 的 `const` 是 `attribute.Key` 型別，有 `.String()` / `.Int()` / `.Float64()` / `.Bool()` 方法：

```go
import "myservice/generated_from_template"  // package semconv

span.SetAttributes(
    semconv.PAYMENT_ORDER_ID.String("ord-001"),    // → "payment.order_id" = "ord-001"
    semconv.CART_ITEM_QUANTITY.Int(3),              // → "cart.item_quantity" = 3
    semconv.CART_ITEM_PRICE.Float64(299.0),         // → "cart.item_price" = 299.0
    semconv.IS_RETRY.Bool(false),                   // → "is_retry" = false
)
```

Python 的常數就是字串，直接傳給 `set_attribute`：

```python
from generated_from_template.semconv_attrs import PAYMENT_ORDER_ID, CART_ITEM_QUANTITY

span.set_attribute(PAYMENT_ORDER_ID, "ord-001")
span.set_attribute(CART_ITEM_QUANTITY, 3)
```

### `semconv_metrics` — Metric 初始化三件組

每個 metric 生成三個常數 `_NAME` / `_UNIT` / `_DESC`，專門用來建立 metric instrument：

```go
// Go
hist, _ := meter.Float64Histogram(
    semconv.PAYMENT_AMOUNT_NAME,
    metric.WithUnit(semconv.PAYMENT_AMOUNT_UNIT),
    metric.WithDescription(semconv.PAYMENT_AMOUNT_DESC),
)
```

```python
# Python
hist = meter.create_histogram(
    PAYMENT_AMOUNT_NAME,
    unit=PAYMENT_AMOUNT_UNIT,
    description=PAYMENT_AMOUNT_DESC,
)
```

## 6.4 完整 Go 業務邏輯整合範例

```go
// internal/payment/service.go
package payment

import (
    "context"
    "fmt"

    "go.opentelemetry.io/otel"
    otelmetric "go.opentelemetry.io/otel/metric"
    "go.opentelemetry.io/otel/trace"
    semconv "myservice/generated_from_template"
)

var tracer = otel.Tracer("payment-service")

// PaymentService 封裝支付業務邏輯與遙測 instruments
type PaymentService struct {
    amountHisto  otelmetric.Float64Histogram
    errorCounter otelmetric.Int64Counter
}

// NewPaymentService 建立 PaymentService，初始化 metric instruments
func NewPaymentService() (*PaymentService, error) {
    meter := otel.Meter("payment-service")

    // 使用生成的常數，確保名稱、單位、描述與 Schema 一致
    hist, err := meter.Float64Histogram(
        semconv.PAYMENT_AMOUNT_NAME,
        otelmetric.WithUnit(semconv.PAYMENT_AMOUNT_UNIT),
        otelmetric.WithDescription(semconv.PAYMENT_AMOUNT_DESC),
    )
    if err != nil {
        return nil, fmt.Errorf("建立 payment amount histogram 失敗: %w", err)
    }

    counter, err := meter.Int64Counter(
        semconv.PAYMENT_ERRORS_NAME,
        otelmetric.WithUnit(semconv.PAYMENT_ERRORS_UNIT),
        otelmetric.WithDescription(semconv.PAYMENT_ERRORS_DESC),
    )
    if err != nil {
        return nil, fmt.Errorf("建立 payment errors counter 失敗: %w", err)
    }

    return &PaymentService{
        amountHisto:  hist,
        errorCounter: counter,
    }, nil
}

type PaymentRequest struct {
    OrderID  string
    Amount   float64
    Provider string
    GitTag   string
    Env      string
}

// ProcessPayment 處理支付，同時發出符合 Schema 的遙測訊號
func (s *PaymentService) ProcessPayment(ctx context.Context, req PaymentRequest) error {
    ctx, span := tracer.Start(ctx, "payment.process",
        trace.WithSpanKind(trace.SpanKindServer),
    )
    defer span.End()

    // ✓ 使用生成的常數，IDE 自動補齊，編譯期保證屬性名稱正確
    span.SetAttributes(
        semconv.PAYMENT_ORDER_ID.String(req.OrderID),
        semconv.PAYMENT_PROVIDER.String(req.Provider),
        semconv.GIT_TAG.String(req.GitTag),
        semconv.DEPLOYMENT_ENVIRONMENT.String(req.Env),
    )

    // 執行支付邏輯（省略實際實作）
    if req.Amount <= 0 {
        errType := "invalid_amount"
        // conditionally_required：只在失敗時設定 error.type
        span.SetAttributes(semconv.ERROR_TYPE.String(errType))
        span.SetAttributes(semconv.PAYMENT_STATUS.String("failed"))

        s.errorCounter.Add(ctx, 1,
            otelmetric.WithAttributes(semconv.ERROR_TYPE.String(errType)),
        )
        return fmt.Errorf("金額無效: %f", req.Amount)
    }

    span.SetAttributes(semconv.PAYMENT_STATUS.String("success"))
    s.amountHisto.Record(ctx, req.Amount,
        otelmetric.WithAttributes(
            semconv.PAYMENT_PROVIDER.String(req.Provider),
            semconv.DEPLOYMENT_ENVIRONMENT.String(req.Env),
        ),
    )

    return nil
}
```

## 6.6 完整 Python 業務邏輯整合範例

```python
# payment/service.py
from __future__ import annotations

from opentelemetry import trace, metrics
from opentelemetry.sdk.trace import TracerProvider
from opentelemetry.sdk.metrics import MeterProvider

# 從生成的常數 import
from generated_from_template.semconv_attrs import (
    PAYMENT_ORDER_ID,
    PAYMENT_PROVIDER,
    PAYMENT_STATUS,
    ERROR_TYPE,
    GIT_TAG,
    DEPLOYMENT_ENVIRONMENT,
)
from generated_from_template.semconv_metrics import (
    PAYMENT_AMOUNT_NAME,
    PAYMENT_AMOUNT_UNIT,
    PAYMENT_AMOUNT_DESC,
    PAYMENT_ERRORS_NAME,
    PAYMENT_ERRORS_UNIT,
    PAYMENT_ERRORS_DESC,
)

tracer = trace.get_tracer("payment-service")
meter = metrics.get_meter("payment-service")

# 使用生成的常數建立 metric instruments，確保與 Schema 一致
_amount_histo = meter.create_histogram(
    name=PAYMENT_AMOUNT_NAME,
    unit=PAYMENT_AMOUNT_UNIT,
    description=PAYMENT_AMOUNT_DESC,
)

_error_counter = meter.create_counter(
    name=PAYMENT_ERRORS_NAME,
    unit=PAYMENT_ERRORS_UNIT,
    description=PAYMENT_ERRORS_DESC,
)


def process_payment(
    order_id: str,
    provider: str,
    amount: float,
    git_tag: str = "unknown",
    env: str = "production",
) -> bool:
    """
    處理支付，同時發出符合 Schema 的遙測訊號。

    Returns:
        True 表示支付成功，False 表示失敗。
    """
    with tracer.start_as_current_span("payment.process") as span:
        # 使用常數，任何拼字錯誤都在 import 時就會失敗
        span.set_attributes({
            PAYMENT_ORDER_ID: order_id,
            PAYMENT_PROVIDER: provider,
            GIT_TAG: git_tag,
            DEPLOYMENT_ENVIRONMENT: env,
        })

        if amount <= 0:
            error_type = "invalid_amount"
            # conditionally_required：只在失敗時設定
            span.set_attributes({
                ERROR_TYPE: error_type,
                PAYMENT_STATUS: "failed",
            })
            _error_counter.add(
                1,
                attributes={ERROR_TYPE: error_type},
            )
            return False

        span.set_attribute(PAYMENT_STATUS, "success")
        _amount_histo.record(
            amount,
            attributes={
                PAYMENT_PROVIDER: provider,
                DEPLOYMENT_ENVIRONMENT: env,
            },
        )
        return True
```

## 6.9 常見錯誤對照

### 錯誤一：import 路徑錯誤

```go
// ❌ 錯誤：package 路徑不對
import "semconv"

// ✓ 正確：使用完整的 module 路徑
import semconv "myservice/generated_from_template"
```

```python
# ❌ 錯誤：相對 import 可能在某些執行環境下失敗
from .semconv_attrs import PAYMENT_ORDER_ID

# ✓ 正確：使用絕對 import（從 module 根目錄）
from generated_from_template.semconv_attrs import PAYMENT_ORDER_ID
```

### 錯誤二：使用 attribute.Key 時型別不符

```go
// ❌ 錯誤：CART_ITEM_QUANTITY 是整數型別，不能用 .String()
span.SetAttributes(
    semconv.CART_ITEM_QUANTITY.String("3"),  // 型別錯誤！
)

// ✓ 正確：依照 Schema 中 type=int，使用 .Int()
span.SetAttributes(
    semconv.CART_ITEM_QUANTITY.Int(3),
)
```

### 錯誤三：在 Python 中把常數當 key 物件用

```python
# ❌ 錯誤：Python 的常數是字串，不是 key 物件
span.set_attribute(PAYMENT_ORDER_ID, ...)    # 正確
span.set_attribute(PAYMENT_ORDER_ID.str, ...)  # AttributeError！

# ✓ 正確：直接使用字串常數
span.set_attribute(PAYMENT_ORDER_ID, order_id)
```
